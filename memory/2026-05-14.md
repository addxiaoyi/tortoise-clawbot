## 核心特性

### Agent Engine
```rust
// 创建代理
let agent = AgentFactory::create_openai("Tortoise", api_key, "gpt-4").await?;

// 5种思维模式
ThinkMode::auto_select("深度分析这个问题"); // -> Deep
ThinkMode::Creative; // 创意模式

// 聊天
let response = agent.chat(messages, options).await?;
```

### Memory System
```rust
// 三层记忆
memory.remember("用户偏好", 0.9).await?;  // 高重要性直接进长期
let memories = memory.recall("查询相关记忆").await?;

// 自动遗忘
memory.forget_low_importance(0.3).await?;
```

### Channel System
```rust
// 统一消息格式
let msg = UnifiedMessage { ... };

// 路由器
router.register_channel(discord).await;
router.route(msg).await?;
```

### Plugin System
```rust
// 插件管理
manager.load_plugin(path).await?;
manager.enable_plugin("discord").await?;

// WASM 沙盒
Sandbox::new(128 * 1024 * 1024)?; // 128MB 限制
```
